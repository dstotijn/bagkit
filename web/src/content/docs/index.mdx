import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';
import { withBase } from '../../base';

<CardGrid>
  <Card title="Reads the national BAG extract" icon="document">
    Give it the [monthly BAG extract](https://www.kadaster.nl/-/gratis-download-bag-extract) from
    PDOK as‑is, or the XML files from it. bagkit keeps the voorkomens that are valid on a date you
    choose, and checks identificaties and references as it goes.
  </Card>
  <Card title="Writes SQLite, Parquet or CSV" icon="seti:db">
    One table per object type, or one row per address with its names and coordinates. Convert only
    the object types you need. SQLite output is a GeoPackage, Parquet is GeoParquet.
  </Card>
  <Card title="Looks up addresses" icon="magnifier">
    Find an address by postcode and house number, or the addresses nearest to a coordinate, with
    `bagkit query` or over HTTP with `bagkit serve`.
  </Card>
  <Card title="Exposes a Rust library" icon="seti:rust">
    The `bagkit` command is built on a crate you can use directly: stream typed voorkomens into your
    own code, or write any of the outputs from it.
  </Card>
</CardGrid>
